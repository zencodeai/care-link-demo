    "assistance.actor_resumption_required",
    "assistance.channel_changed",
    "assistance.function_disabled",

    "manual.workload_denied",
}
DATA_DIAGNOSTIC_OPERATIONS = {
    "/internal/v1/data/task_create": "task_create",
    "/internal/v1/data/assistance/inference/context": "inference_context",
    "/internal/v1/data/assistance/inference/current": "inference_current",
    "/internal/v1/data/assistance/inference/reserve": "inference_reserve",
}


def observe_task_denials(app: ASGIApp) -> ASGIApp:
    """Keep only fixed task and inference error categories from this synthetic role."""

    async def observed(scope: Scope, receive: Receive, send: Send) -> None:
        if scope["type"] != "http":
            await app(scope, receive, send)
            return
        operation = DATA_DIAGNOSTIC_OPERATIONS.get(scope["path"])
        if operation is None:
            await app(scope, receive, send)
            return
        failed = False

        async def report(message: Message) -> None:
            nonlocal failed
            if message["type"] == "http.response.start":
                failed = message["status"] >= 400
            if failed and message["type"] == "http.response.body":
                body = message.get("body", b"")
                if 0 < len(body) <= 4096 and not message.get("more_body"):
                    try:
                        key = json.loads(body).get("message_key")
                    except (UnicodeDecodeError, ValueError, AttributeError):
                        key = None
                    print(
                        f"IMP10 Data denial operation={operation}: "
                        + (key if isinstance(key, str) and key in TASK_DENIALS else "other"),
                        flush=True,
                    )
            await send(message)

        await app(scope, receive, report)

    return observed


def observe_broker_failure(app: ASGIApp) -> ASGIApp:
    """Observe exception class and product frame for one isolated broker operation."""

    async def observed(scope: Scope, receive: Receive, send: Send) -> None:
        try:
            await app(scope, receive, send)
        except Exception as error:
            if scope["type"] == "http" and scope["path"] == "/internal/v1/broker/inference":
                frames = [
                    {
                        "module": Path(frame.filename).name,
                        "line": frame.lineno,
                        "function": frame.name,
                    }
                    for frame in extract_tb(error.__traceback__)
                    if "/care_link_" in frame.filename
                ]
                print(
                    "IMP10 broker failure: "
                    + json.dumps({"type": type(error).__name__, "frames": frames[-5:]}),
                    flush=True,
                )
            raise

    return observed


def run(configuration: Path) -> None:
    """Bind a role only to the disposable qualification ledger, never serving trust."""
    cfg = load(configuration)
    role = text(cfg, "role")
    uid = integer(cfg, "uid")
    owner = integer(cfg, "release_owner")
    if role not in ROLES or cfg.get("session") != "imp10-protected" or uid != os.geteuid():
        raise ValueError("Exact isolated qualification role required")
    if uid == 0 or owner in {uid, 0}:
        raise ValueError("Distinct release and application custody required")
    shared = Path(text(cfg, "shared_release"))
    observed = ReleaseObservation.model_validate_json(
        Path(text(cfg, "role_observation")).read_bytes()
    )
    release = load_qualification_bindings(
        shared,
        owner_uid=owner,
        role=role,
        boot_id=text(cfg, "boot_id"),
        expected_observation=observed,
    )
    release.release("worker_summary")
    if role == "runtime":
        socket = Path(text(cfg, "core_socket"))
        marker = Path("/state/dispatch-pair")
        while True:
            if not release.current() or not marker.exists():
                sleep(0.1)
                continue
            marker.unlink()
            with ThreadPoolExecutor(max_workers=2) as workers:
                running = [
                    workers.submit(start_runtime, release, core_socket=socket, once=True)
                    for _ in range(2)
                ]
                for worker in running:
                    worker.result()
        return

    proxy, fence_uid = integer(cfg, "proxy_uid"), integer(cfg, "fence_uid")
    if len({uid, proxy, fence_uid, owner}) != 4 or 0 in {proxy, fence_uid}:
        raise ValueError("Distinct protected inference custody required")
    fence_socket, backend = Path(text(cfg, "fence_socket")), Path(text(cfg, "backend_socket"))
    verifier = backend_verifier(
        cast(BackendRole, role),
        proxy_uid=proxy,
        fence_socket=fence_socket,
        fence_uid=fence_uid,
        training_enabled=cfg.get("training_enabled") is True,
    )
    if role == "inference":
        start_inference(
            release,
            verifier,
            backend_socket=backend,
            data_socket=Path(text(cfg, "data_socket")),
            bridge_socket=Path(text(cfg, "bridge_socket")),
        )
    elif role == "host-inference-bridge":
        start_host_bridge(
            release,
            verifier,
            backend_socket=backend,
            artifact_directory=Path(text(cfg, "artifact_directory")),
            model_index=Path(text(cfg, "model_index")),
            lmstudio_token_file=Path(text(cfg, "lmstudio_token_file")),
        )
    else:
        fences = UnixFenceClient(fence_socket, fence_uid)
        knowledge = load_knowledge_resources()
        options = (
            {"inference_socket": Path(text(cfg, "inference_socket"))}
            if role == "core"
            else {
                "provider_stop": ProviderStopAuthority(
                    Path(text(cfg, "stop_receipts")),
                    public_key(text(cfg, "stop_public_key")),
                    fences,
                    issuer=text(cfg, "stop_issuer"),
                    target=TARGET,
                    owner_uid=integer(cfg, "stop_owner_uid"),
                    current_generation=text(cfg, "bridge_generation"),
                )
            }
        )
        application = Composition(
            cfg, ai=ManualRoleAssistance(release, verifier, fences, knowledge, **options)
        )
        selected = (
            observe_task_denials(application)
            if role == "data"
            else observe_broker_failure(application)
        )
        serve_backend(selected, backend_socket=backend, proxy_uid=proxy)
